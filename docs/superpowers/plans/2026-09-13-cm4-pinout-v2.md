```markdown
# ADV7280A-M decoder on the CM4 baseboard (from sdtv2csi)

Source: github.com/rniwase/sdtv2csi (KiCad 9 schematic + PCB, MIT). Board photo: sdtv2csi/misc/*.jpg.

## What the reference board contains
- U1 ADV7280A-M (Analog Devices SDTV video decoder, MIPI CSI-2 output, 1 data lane).
- U2 NJM12888F18 LDO: 3.3 V in → 1.8 V for AVDD/DVDD/MVDD/PVDD (each rail through a 600 Ω ferrite bead, 0.1 µF + 10 µF per rail). DVDDIO = 3.3 V.
- Y1 ABM8G 28.63636 MHz crystal, 2 × 20 pF.
- 8 analogue inputs AIN1–AIN8: each VINx → 24 Ω series → 0.1 µF (AC coupling) → AINx, 51 Ω to GND at the input (termination). Inputs come from a 2×8 pin header (VIDEO_IN).
- CSI-2: D0P/D0N (U1 pins 9/10) and CLKP/CLKN (11/12) to the 15-pin Raspberry Pi camera FFC; D1 pins on the FFC left unconnected.
- I2C: SDA/SCL (U1 30/31) to FFC pins 2/3 (= CM4 SDA0/SCL0). ALSB (29) via 4.7 kΩ pull-up (R20) with DNP pull-down (R21): ALSB high → I2C address 0x21 (7-bit), low → 0x20.
- ~RESET (28): 4.7 kΩ pull-up + 4.7 µF to GND (RC power-on reset). ~PWRDWN (32): 4.7 kΩ pull-up. ~INTRQ, GPO0–GPO2: test points only.
- FFC pins 4/5 ("GPIO0/GPIO1" nets): test points only, not used by the decoder.

## Software (Raspberry Pi OS)
- `dtoverlay=adv728x-m,adv7280m=1` (default I2C address 0x21; `addr=0x20` if ALSB is low). The overlay targets CAM1 = CSI1 with I2C on `i2c_csi_dsi` (CM4 SDA0/SCL0, pins 82/80). Kernel driver adv7180 (patched for 507-line NTSC frames in sdtv2csi/rpi/install_patch.sh).

## Consequences for the CM4 pinout (decided 2026-09-13)
- CSI1 D0 (pins 115/117) and CSI1 clock (127/129) → ADV7280A-M. CSI1 D1–D3 unused → leave unconnected.
- I2C0 (pins 82/80) shared by the CSI0 camera and the decoder (0x20/0x21). A CSI0 camera must use a custom overlay variant with `i2c_csi_dsi` (stock `cam0` variants expect I2C on GPIO0/1, which are UART2 on this board).
- No CM4 GPIO for the decoder: reset by RC, power-down pulled up (as on sdtv2csi). CVBS switch + K7 encoder path is kept in parallel.
```

```python
import json, re, unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
STATUSES = {"used", "opt", "free", "nc", "dnc"}
CYR = re.compile(r"[Ѐ-ӿ]")


class TestProject(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.ds = json.loads((ROOT / "data" / "cm4_datasheet.json").read_text(encoding="utf-8"))
        cls.prj = json.loads((ROOT / "data" / "project.json").read_text(encoding="utf-8"))
        cls.pins = {p["n"]: p for p in cls.ds["pins"]}

    def entry(self, n):
        p = self.pins[n]
        e = self.prj["pins"].get(str(n)) or self.prj["defaults"].get(p["grp"]) or {}
        return {"net": e.get("net", ""), "dev": e.get("dev", "NC"), "mode": e.get("mode", ""),
                "status": e.get("status", "used" if str(n) in self.prj["pins"] else "free"), "note": e.get("note", "")}

    def test_shape_v2(self):
        for key in ("meta", "devices", "defaults", "pins"):
            self.assertIn(key, self.prj)
        for key in ("ds_notes", "facts"):
            self.assertNotIn(key, self.prj)
        for k, d in self.prj["devices"].items():
            self.assertTrue(d["name"] and re.fullmatch(r"#[0-9A-Fa-f]{6}", d["c"]), k)
        self.assertIn("NC", self.prj["devices"])
        self.assertIn("VDEC", self.prj["devices"])

    def test_english_only(self):
        blob = json.dumps(self.prj, ensure_ascii=False)
        self.assertIsNone(CYR.search(blob), "Cyrillic text left in project.json")

    def test_pins_reference_existing(self):
        for k, e in self.prj["pins"].items():
            self.assertIn(int(k), self.pins, k)
            self.assertIn(e["dev"], self.prj["devices"], k)
            self.assertIn(e.get("status", "used"), STATUSES, k)
            self.assertTrue(e.get("note"), f"pin {k} has no note")

                continue
            e = self.prj["pins"].get(str(n))

            self.assertNotIn(p["gpio"], seen)
            seen[p["gpio"]] = n
            m = re.fullmatch(r"ALT(\d) (.+)", e["mode"])
            if m:
                self.assertEqual(self.ds["alt"][str(p["gpio"])]["alt"][int(m.group(1))], m.group(2), f"pin {n}")
            else:
                self.assertEqual(e["mode"], "GPIO", f"pin {n}")
        self.assertEqual(sorted(seen), list(range(28)))

    def test_nets_unique_outside_power(self):
        nets = {}
        for n in range(1, 201):
            e = self.entry(n)
            if not e["net"] or self.pins[n]["grp"] in ("GND", "PWR5", "PWR33"):
                continue
            self.assertNotIn(e["net"], nets, f"net {e['net']}: pins {nets.get(e['net'])} and {n}")
            nets[e["net"]] = n

    def test_diff_pair_polarity_and_modes(self):
```

```js
  };
  var DIRS = {gnd:'ground', power_in:'power input', power_out:'power output', in:'input', out:'output', bidir:'bidirectional', nc:'do not connect'};
  var STATUS = {used:'used', opt:'optional / reserved', free:'free', nc:'not used', dnc:'do not connect'};
  var PITCH = 18, PAD = 16, X0 = 60, Y = {J1:{odd:34, even:54}, J2:{odd:484, even:504}};
  var MM = 8.4, OX = 269, OY = 118;   // outline scale (px per mm) and origin of the 55×40 mm board

  function padXY(n) {
    var k = (n - 1) % 100, col = Math.floor(k / 2);
    return {x: X0 + (49 - col) * PITCH, y: Y[n <= 100 ? 'J1' : 'J2'][n % 2 ? 'odd' : 'even']};
  }
  function mm(x, y) { return {x: OX + x * MM, y: OY + y * MM}; }
  function merge(pin, prj) {
    var explicit = prj.pins[String(pin.n)], e = explicit || prj.defaults[pin.grp] || {};
    var dev = e.dev || 'NC', d = prj.devices[dev];
    return {net: e.net || '', dev: dev, devName: d ? d.name : '', mode: e.mode || '',
            status: e.status || (explicit ? 'used' : 'free'), note: e.note || ''};
  }
  function colorOf(pin, ov, layer, prj) {
    if (layer === 'ds') return GROUPS[pin.grp] ? GROUPS[pin.grp].c : '#8C8C8C';
    if (ov.status === 'dnc') return '#B22222';
    if (ov.status === 'nc' || ov.status === 'free') return '#9A9A9A';
```
